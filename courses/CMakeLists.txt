file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/manifest.txt" course_entries)
set(course_files)
foreach(entry IN LISTS course_entries)
    if(entry MATCHES "^#" OR entry STREQUAL "")
        continue()
    endif()
    string(REPLACE "|" ";" fields "${entry}")
    list(LENGTH fields field_count)
    if(NOT field_count EQUAL 2)
        message(FATAL_ERROR "Invalid course manifest entry: ${entry}")
    endif()
    list(GET fields 1 file_name)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${file_name}")
        message(FATAL_ERROR "Missing shipped course: ${file_name}")
    endif()
    list(APPEND course_files "${file_name}")
endforeach()
install(FILES ${course_files} manifest.txt DESTINATION ${KDE_INSTALL_DATADIR}/kolf/courses)
